# Documentation site toolchain.
#
# Fully pinned, including transitive packages that affect rendering, so a local
# build and a CI build produce identical output.
#
# Install:  pip install -r requirements.txt
# Build:    python3 scripts/build_manifest.py && mkdocs build -f mkdocs.generated.yml --strict
# Preview:  python3 scripts/build_manifest.py && mkdocs serve -f mkdocs.generated.yml
#
# The manifest generator (scripts/build_manifest.py) itself needs no
# dependencies - it is standard library only, so the repo's generated content
# can be refreshed without installing anything here.
#
# Two pins are load-bearing rather than merely tidy:
#
#   pymdown-extensions >= 11 with Pygments 2.20. Under pymdown-extensions 10.x,
#   Pygments 2.20 raises `AttributeError: 'NoneType' object has no attribute
#   'replace'` on any indented code block, which aborts the build.
#
#   mkdocs-material 9.x. The Material team has said MkDocs 2.0 removes the
#   plugin system and rewrites theming with no migration path, so an unpinned
#   upgrade would break this build outright. Bump deliberately and rebuild with
#   --strict.

mkdocs==1.6.1
mkdocs-material==9.7.7
mkdocs-material-extensions==1.3.1
mkdocs-minify-plugin==0.8.0
pymdown-extensions==11.0.1
Pygments==2.20.0
Markdown==3.10.3
